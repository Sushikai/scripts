### 8. `_btMaybeAutoCompare` 自动对比 (line ~3667)
```javascript
function _btMaybeAutoCompare(final) {
  const curSig = _btParamsSig(window._BT_LAST_BODY || {});
  const curStrat = final?.config?.strategy_id || 'baseline';
  window._BT_RESULT_SIG = window._BT_RESULT_SIG || {};
  window._BT_RESULT_SIG[curStrat] = curSig;
  const other = curStrat === 'WIN_RATE_1000' ? 'baseline' : 'WIN_RATE_1000';
  const otherReady = (window._BT_RESULTS?.[other]) && (window._BT_RESULT_SIG[other] === curSig);
  if (otherReady) {
    window._BT_AUTO_COMPARE_RUNNING = false;
    btRenderCompare(curSig);
    return;
  }
  if (!window._BT_AUTO_COMPARE_RUNNING) {
    // 启动一次对手策略自动跑 (同参数)
    window._BT_AUTO_COMPARE_RUNNING = true;
    window._BT_STRATEGY_ID = other;
    const otherLabel = other === 'WIN_RATE_1000' ? '🔥高胜率' : '基线';
    toast(`自动对比: 用相同参数继续跑 ${otherLabel} 方案…`, 2600);
    setTimeout(() => btStart(), 500);
  } else {
    // 自动第二轮已跑完但仍未匹配 (参数漂移) → 停止链, 尝试渲染现有
    window._BT_AUTO_COMPARE_RUNNING = false;
    btRenderCompare(curSig);
  }
}
```
